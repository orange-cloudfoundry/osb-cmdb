1. Get the application URL by running these commands:
{{- if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "osb-cmdb-go.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "osb-cmdb-go.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "osb-cmdb-go.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "osb-cmdb-go.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. Verify the broker is healthy:
  curl -u {{ .Values.broker.username }}:<password> http://<broker-url>/v2/catalog

3. Register the broker with your platform:
  # For CF:
  cf create-service-broker osb-cmdb-go {{ .Values.broker.username }} <password> http://<broker-url>

  # For K8S Service Catalog:
  kubectl apply -f - <<EOF
  apiVersion: servicecatalog.k8s.io/v1beta1
  kind: ClusterServiceBroker
  metadata:
    name: osb-cmdb-go
  spec:
    url: http://{{ include "osb-cmdb-go.fullname" . }}.{{ .Release.Namespace }}.svc:{{ .Values.service.port }}
    authInfo:
      basic:
        secretRef:
          namespace: {{ .Release.Namespace }}
          name: {{ include "osb-cmdb-go.secretName" . }}
  EOF

For more information, visit: https://github.com/orange-cloudfoundry/osb-cmdb-go
